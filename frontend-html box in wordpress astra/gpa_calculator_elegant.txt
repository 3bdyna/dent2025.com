<script>
    // Global State for GPA Calculator
    let dentGpaSubjects = [];
    let dentGpaSelection = null;
    let dentGpaGradeMode = 'letter'; // 'letter' or 'marks'

    function parseLocalNum(val) {
        if (val === null || val === undefined) return 0;
        const s = String(val).replace(/[٠-٩]/g, d => "٠١٢٣٤٥٦٧٨٩".indexOf(d)).replace(/[^\d.-]/g, '');
        const num = parseFloat(s);
        return isNaN(num) ? 0 : num;
    }

    const dentGradePoints5 = {
        "": 0,
        "A+": 5.0,
        "A": 4.75,
        "B+": 4.5,
        "B": 4.0,
        "C+": 3.5,
        "C": 3.0,
        "D+": 2.5,
        "D": 2.0,
        "F": 0.0
    };

    function marksToGrade(marks) {
        const val = parseLocalNum(marks);
        if (val <= 0 && String(marks).trim() === '') return "";
        if (val >= 95) return "A+";
        if (val >= 90) return "A";
        if (val >= 85) return "B+";
        if (val >= 80) return "B";
        if (val >= 75) return "C+";
        if (val >= 70) return "C";
        if (val >= 65) return "D+";
        if (val >= 60) return "D";
        return "F";
    }

    // Modal Control
    function openGPAModal() {
        const modal = document.getElementById('gpaModal');
        if (modal) {
            if (modal.parentNode !== document.body) {
                document.body.appendChild(modal);
            }
            modal.style.setProperty('display', 'flex', 'important');
            // Trigger reflow before adding active for smooth opacity transition
            void modal.offsetWidth;
            modal.classList.add('active');
            document.body.style.overflow = 'hidden';
            initDentGpaCalculator();
        }
    }

    function closeGPAModal() {
        const modal = document.getElementById('gpaModal');
        if (modal) {
            modal.classList.remove('active');
            document.body.style.overflow = '';
            setTimeout(function() {
                if (!modal.classList.contains('active')) {
                    modal.style.setProperty('display', 'none', 'important');
                }
            }, 250);
        }
    }

    function closeModalOnBgClick(e) {
        if (e.target.id === 'gpaModal') {
            closeGPAModal();
        }
    }

    function initDentGpaCalculator() {
        const rawSel = localStorage.getItem('dent2025_selection');
        if (rawSel) {
            try {
                dentGpaSelection = JSON.parse(rawSel);
            } catch (e) {
                dentGpaSelection = null;
            }
        } else {
            dentGpaSelection = null;
        }

        updateGpaHeaderTitle();
        loadDentGpaSubjects();
    }

    function updateGpaHeaderTitle() {
        const titleEl = document.getElementById('gpa-modal-subtitle');
        if (!titleEl) return;

        if (dentGpaSelection) {
            let specText = dentGpaSelection.specialty === 'dentistry' ? 'طب الأسنان' : (dentGpaSelection.specialty === 'medicine' ? 'الطب البشري' : 'التحضيري');
            let semText = `السنة ${dentGpaSelection.year} - الفصل ${dentGpaSelection.semester}`;
            if (dentGpaSelection.specialty === 'pre-med') {
                semText = `التحضيري - الفصل ${dentGpaSelection.semester}`;
            }
            titleEl.innerText = `${specText} (${semText})`;
        } else {
            titleEl.innerText = 'حاسبة المعدل الأكاديمي';
        }
    }

    function getStorageKey() {
        if (dentGpaSelection) {
            return `dent2025_gpa_${dentGpaSelection.specialty}_y${dentGpaSelection.year}_s${dentGpaSelection.semester}`;
        }
        return 'dent2025_simple_gpa';
    }

    function loadDentGpaSubjects() {
        let apiSubjects = [];

        if (dentGpaSelection) {
            const cacheKey = `dent2025_dashboard_data_${dentGpaSelection.specialty}_${dentGpaSelection.year}_${dentGpaSelection.semester}`;
            const cached = sessionStorage.getItem(cacheKey);
            if (cached) {
                try {
                    const parsed = JSON.parse(cached);
                    if (parsed.subjects) {
                        apiSubjects = parsed.subjects;
                    }
                } catch(e){}
            }
        }

        // If no cached subjects, attempt fetch from API
        if (dentGpaSelection) {
            if (apiSubjects.length === 0) {
                const apiBase = (window.location.pathname === '/dev' || window.location.pathname.startsWith('/dev/')) ? '/dev' : '';
                fetch(`${apiBase}/dent2025_api.php?action=data&specialty=${dentGpaSelection.specialty}&year=${dentGpaSelection.year}&semester=${dentGpaSelection.semester}`)
                    .then(res => res.json())
                    .then(resData => {
                        if (resData.success) {
                            if (resData.data) {
                                if (resData.data.subjects) {
                                    apiSubjects = resData.data.subjects;
                                    buildSubjectsList(apiSubjects);
                                }
                            }
                        }
                    })
                    .catch(err => {
                        buildSubjectsList([]);
                    });
                return;
            }
        }

        buildSubjectsList(apiSubjects);
    }

    function buildSubjectsList(apiSubjects) {
        dentGpaSubjects = [];

        // Check local storage for saved state
        const storageKey = getStorageKey();
        const savedRaw = localStorage.getItem(storageKey);
        let savedState = null;
        if (savedRaw) {
            try { savedState = JSON.parse(savedRaw); } catch(e){}
        }

        // Check if old default subjects need fallback if no selection and no API
        if (apiSubjects.length === 0) {
            if (!dentGpaSelection) {
                apiSubjects = [
                    { id: "od", name: "تشخيص الفم", hours: 2 },
                    { id: "gp", name: "علم الأمراض العام", hours: 3 },
                    { id: "abm", name: "مواد طب الأسنان المتقدمة", hours: 2 },
                    { id: "omi", name: "الأحياء الدقيقة", hours: 3 },
                    { id: "dph", name: "الصحة العامة", hours: 1 }
                ];
            }
        }

        // Map API subjects to local state
        apiSubjects.forEach((sub, idx) => {
            let parsedHours = 0;
            if (typeof sub.hours === 'string') {
                const tMatch = sub.hours.match(/(\d+)\s*نظري/);
                const pMatch = sub.hours.match(/(\d+)\s*عملي/);
                let th = tMatch ? parseFloat(tMatch[1]) : 0;
                let pr = pMatch ? parseFloat(pMatch[1]) : 0;
                if (th > 0 || pr > 0) {
                    parsedHours = th + pr;
                } else {
                    parsedHours = parseFloat(sub.hours) || 0;
                }
            } else {
                parsedHours = parseFloat(sub.hours) || 0;
            }

            let isMissing = (isNaN(parsedHours) || parsedHours <= 0);
            if (isMissing) parsedHours = 0;

            let restoredHours = parsedHours;
            let restoredGrade = "";
            let restoredMarks = "";

            if (savedState) {
                if (savedState.grades) {
                    const savedItem = savedState.grades[sub.id || idx];
                    if (savedItem) {
                        if (savedItem.grade !== undefined) restoredGrade = savedItem.grade;
                        if (savedItem.marks !== undefined) restoredMarks = savedItem.marks;
                        if (savedItem.hours !== undefined) {
                            const sh = parseFloat(savedItem.hours);
                            if (!isNaN(sh)) {
                                if (sh > 0) restoredHours = sh;
                            }
                        }
                    }
                }
            }

            dentGpaSubjects.push({
                id: sub.id || `sub_${idx}`,
                name: sub.name || sub.nameAr || `مادة ${idx + 1}`,
                hours: restoredHours,
                isMissingHours: isMissing,
                isCustom: false,
                grade: restoredGrade,
                marks: restoredMarks
            });
        });

        // Restore custom subjects if saved
        if (savedState) {
            if (savedState.customSubjects) {
                if (Array.isArray(savedState.customSubjects)) {
                    savedState.customSubjects.forEach(cSub => {
                        dentGpaSubjects.push({
                            id: cSub.id || `custom_${Date.now()}_${Math.random()}`,
                            name: cSub.name || 'مادة إضافية',
                            hours: parseFloat(cSub.hours) || 2,
                            isMissingHours: false,
                            isCustom: true,
                            grade: cSub.grade || '',
                            marks: cSub.marks || ''
                        });
                    });
                }
            }

            if (savedState.prevHours) {
                const phInput = document.getElementById('prev-hours');
                if (phInput) phInput.value = savedState.prevHours;
            }
            if (savedState.prevGpa) {
                const pgInput = document.getElementById('prev-gpa');
                if (pgInput) pgInput.value = savedState.prevGpa;
            }
            if (savedState.targetGpa) {
                const tgInput = document.getElementById('target-gpa');
                if (tgInput) tgInput.value = savedState.targetGpa;
            }
        }

        renderSubjectsTable();
        calculateGPA(false);
    }

    function renderSubjectsTable() {
        const container = document.getElementById('subjects-list');
        if (!container) return;

        if (dentGpaSubjects.length === 0) {
            container.innerHTML = `
                <div style="text-align: center; color: #9ca3af; padding: 18px; font-size: 0.85rem;">
                    لا توجد مواد مسجلة لهذا الفصل حتى الآن.<br>
                    يمكنك إضافة مواد يدوياً باستخدام الزر أدناه.
                </div>
            `;
            return;
        }

        let html = '';
        dentGpaSubjects.forEach((sub, idx) => {
            let hasWarning = false;
            if (sub.isMissingHours) {
                if (sub.hours <= 0) {
                    hasWarning = true;
                }
            }
            const warningBadge = hasWarning ? `<span style="color: #9ca3af; font-size: 0.68rem; display: block;">تحديد الساعات</span>` : '';
            const hoursClass = hasWarning ? 'gpa-hours-input missing-warning' : 'gpa-hours-input';

            let gradeInputHtml = '';
            if (dentGpaGradeMode === 'letter') {
                gradeInputHtml = `
                    <select class="gpa-grade-select" onchange="updateSubjectGrade(${idx}, this.value)">
                        <option value="" ${sub.grade === '' ? 'selected' : ''}>--</option>
                        <option value="A+" ${sub.grade === 'A+' ? 'selected' : ''}>A+</option>
                        <option value="A" ${sub.grade === 'A' ? 'selected' : ''}>A</option>
                        <option value="B+" ${sub.grade === 'B+' ? 'selected' : ''}>B+</option>
                        <option value="B" ${sub.grade === 'B' ? 'selected' : ''}>B</option>
                        <option value="C+" ${sub.grade === 'C+' ? 'selected' : ''}>C+</option>
                        <option value="C" ${sub.grade === 'C' ? 'selected' : ''}>C</option>
                        <option value="D+" ${sub.grade === 'D+' ? 'selected' : ''}>D+</option>
                        <option value="D" ${sub.grade === 'D' ? 'selected' : ''}>D</option>
                        <option value="F" ${sub.grade === 'F' ? 'selected' : ''}>F</option>
                    </select>
                `;
            } else {
                gradeInputHtml = `
                    <input type="number" class="gpa-marks-input" min="0" max="100" value="${sub.marks !== undefined ? sub.marks : ''}" placeholder="/100" oninput="updateSubjectMarks(${idx}, this.value)">
                `;
            }

            const deleteBtn = `<button type="button" onclick="removeSubjectRow(${idx})" title="حذف المادة" style="background: rgba(255, 255, 255, 0.05); border: 1px solid rgba(255, 255, 255, 0.1); color: #9ca3af; border-radius: 6px; width: 26px; height: 26px; cursor: pointer; display: flex; align-items: center; justify-content: center; font-size: 0.85rem; padding: 0;">×</button>`;

            const safeSubName = String(sub.name || '').replace(/[&<>"']/g, m => ({ '&': '&amp;', '<': '&lt;', '>': '&gt;', '"': '&quot;', "'": '&#039;' }[m]));
            html += `
                <div class="gpa-subject-row">
                    <div class="gpa-subject-info">
                        ${sub.isCustom ? `<input type="text" value="${safeSubName}" onchange="updateSubjectName(${idx}, this.value)" style="width: 100%; font-size: 0.82rem; padding: 4px 6px; height: 32px;" placeholder="اسم المادة">` : `<h4 class="gpa-subject-name" title="${safeSubName}">${safeSubName}</h4>`}
                        ${warningBadge}
                    </div>
                    <div style="display: flex; align-items: center; gap: 3px;">
                        <input type="number" min="1" max="12" class="${hoursClass}" value="${sub.hours > 0 ? sub.hours : ''}" placeholder="ساعات" oninput="updateSubjectHours(${idx}, this.value)" title="عدد الساعات المعتمدة">
                        <span style="font-size: 0.72rem; color: #9ca3af;">س</span>
                    </div>
                    <div>
                        ${gradeInputHtml}
                    </div>
                    ${deleteBtn}
                </div>
            `;
        });

        container.innerHTML = html;
    }

    function setGradeMode(mode) {
        dentGpaGradeMode = mode;
        const btnLetter = document.getElementById('gpa-mode-letter');
        const btnMarks = document.getElementById('gpa-mode-marks');
        if (btnLetter) {
            if (mode === 'letter') {
                btnLetter.classList.add('active');
            } else {
                btnLetter.classList.remove('active');
            }
        }
        if (btnMarks) {
            if (mode === 'marks') {
                btnMarks.classList.add('active');
            } else {
                btnMarks.classList.remove('active');
            }
        }
        renderSubjectsTable();
    }

    function updateSubjectHours(index, val) {
        const parsed = parseLocalNum(val);
        dentGpaSubjects[index].hours = parsed;
        calculateGPA(true);
    }

    function updateSubjectGrade(index, val) {
        dentGpaSubjects[index].grade = val;
        calculateGPA(true);
    }

    function updateSubjectMarks(index, val) {
        dentGpaSubjects[index].marks = val;
        dentGpaSubjects[index].grade = marksToGrade(val);
        calculateGPA(true);
    }

    function updateSubjectName(index, val) {
        dentGpaSubjects[index].name = val || 'مادة جديدة';
        saveState();
    }

    function addCustomSubject() {
        dentGpaSubjects.push({
            id: `custom_${Date.now()}`,
            name: 'مادة جديدة',
            hours: 2,
            isMissingHours: false,
            isCustom: true,
            grade: '',
            marks: ''
        });
        renderSubjectsTable();
        calculateGPA(true);
    }

    function removeSubjectRow(index) {
        dentGpaSubjects.splice(index, 1);
        renderSubjectsTable();
        calculateGPA(true);
    }

    function clearGpaGrades() {
        dentGpaSubjects.forEach(sub => {
            sub.grade = '';
            sub.marks = '';
        });
        renderSubjectsTable();
        calculateGPA(true);
    }

    function calculateGPA(save = false) {
        let semesterPoints = 0;
        let semesterCredits = 0;

        dentGpaSubjects.forEach(sub => {
            const hrs = parseLocalNum(sub.hours);
            const pts = dentGradePoints5[sub.grade] !== undefined ? dentGradePoints5[sub.grade] : 0;
            if (hrs > 0) {
                if (sub.grade) {
                    semesterPoints += (pts * hrs);
                    semesterCredits += hrs;
                }
            }
        });

        // Compute Semester Total Hours (all listed courses with assigned hours)
        let totalCurrentTermHours = 0;
        dentGpaSubjects.forEach(sub => {
            const h = parseLocalNum(sub.hours);
            if (h > 0) totalCurrentTermHours += h;
        });

        const currentTermHeader = document.getElementById('current-term-hours-text');
        if (currentTermHeader) {
            currentTermHeader.innerText = `مواد الفصل الحالي (${totalCurrentTermHours} ساعة)`;
        }

        const semesterGPA = semesterCredits > 0 ? (semesterPoints / semesterCredits) : 0;
        const semGpaEl = document.getElementById('semester-gpa');
        if (semGpaEl) semGpaEl.innerText = semesterGPA.toFixed(2);

        const prevHoursEl = document.getElementById('prev-hours');
        const prevGpaEl = document.getElementById('prev-gpa');
        const prevHours = prevHoursEl ? parseLocalNum(prevHoursEl.value) : 0;
        const prevGPA = prevGpaEl ? parseLocalNum(prevGpaEl.value) : 0;

        let cumulativeGPA = 0;
        let totalCreditsAll = prevHours + semesterCredits;

        if (prevHours > 0 || semesterCredits > 0) {
            const totalPoints = (prevHours * prevGPA) + semesterPoints;
            if (totalCreditsAll > 0) {
                cumulativeGPA = totalPoints / totalCreditsAll;
            }
            const cumGpaEl = document.getElementById('cumulative-gpa');
            if (cumGpaEl) cumGpaEl.innerText = cumulativeGPA.toFixed(2);
        } else {
            const cumGpaEl = document.getElementById('cumulative-gpa');
            if (cumGpaEl) cumGpaEl.innerText = "0.00";
        }

        calculateTargetGpaEstimate(prevHours, prevGPA, totalCurrentTermHours, cumulativeGPA);

        if (save) saveState();
    }

    function calculateTargetGpaEstimate(prevHours, prevGPA, termHours, currentCumGpa) {
        const targetInput = document.getElementById('target-gpa');
        const resultEl = document.getElementById('target-gpa-result');
        if (!targetInput) return;
        if (!resultEl) return;

        const targetGpa = parseLocalNum(targetInput.value);
        if (isNaN(targetGpa) || targetGpa <= 0) {
            resultEl.innerHTML = '';
            return;
        }

        if (termHours <= 0) {
            resultEl.innerHTML = '<span style="color: #9ca3af;">الرجاء تحديد ساعات المواد للفصل الحالي أولاً.</span>';
            return;
        }

        const totalHours = prevHours + termHours;
        const reqSemGPA = (targetGpa * totalHours - (prevHours * prevGPA)) / termHours;

        if (reqSemGPA > 5.0) {
            resultEl.innerHTML = `<span style="color: #ef4444;">تحتاج معدل فصلي <strong>${reqSemGPA.toFixed(2)}</strong> (أعلى من 5.0 - غير ممكن في فصل واحد).</span>`;
        } else if (reqSemGPA <= 0) {
            resultEl.innerHTML = `<span style="color: #e2e8f0;">هدف المكتسب محقق بالكامل بفضل معدلك السابق!</span>`;
        } else {
            let letterHint = "A+";
            if (reqSemGPA <= 2.0) letterHint = "D";
            else if (reqSemGPA <= 2.5) letterHint = "D+";
            else if (reqSemGPA <= 3.0) letterHint = "C";
            else if (reqSemGPA <= 3.5) letterHint = "C+";
            else if (reqSemGPA <= 4.0) letterHint = "B";
            else if (reqSemGPA <= 4.5) letterHint = "B+";
            else if (reqSemGPA <= 4.75) letterHint = "A";

            resultEl.innerHTML = `تحتاج إلى معدل فصلي <strong style="color: #f8fafc; font-size: 1.05rem;">${reqSemGPA.toFixed(2)}</strong> (متوسط تقدير <strong>${letterHint}</strong>) لتحقيق هدفك.`;
        }
    }

    function saveState() {
        const prevHoursEl = document.getElementById('prev-hours');
        const prevGpaEl = document.getElementById('prev-gpa');
        const targetGpaEl = document.getElementById('target-gpa');

        const state = {
            prevHours: prevHoursEl ? prevHoursEl.value : '',
            prevGpa: prevGpaEl ? prevGpaEl.value : '',
            targetGpa: targetGpaEl ? targetGpaEl.value : '',
            grades: {},
            customSubjects: []
        };

        dentGpaSubjects.forEach(sub => {
            if (sub.isCustom) {
                state.customSubjects.push(sub);
            } else {
                state.grades[sub.id] = {
                    grade: sub.grade,
                    marks: sub.marks,
                    hours: sub.hours
                };
            }
        });

        const storageKey = getStorageKey();
        try {
            localStorage.setItem(storageKey, JSON.stringify(state));
        } catch(e){}
    }
